# Build and run the C and C++ examples against the wickra-copilot C ABI.
#
#   cargo build --release -p wickra-copilot-c
#   cmake -S examples/c -B examples/c/build
#   cmake --build examples/c/build --config Release
#   ctest --test-dir examples/c/build -C Release --output-on-failure
#
# WICKRA_COPILOT_LIB_DIR defaults to the workspace target/release directory and can
# be overridden for out-of-tree builds.

cmake_minimum_required(VERSION 3.15)
project(wickra_copilot_examples C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(WICKRA_COPILOT_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target/release"
    CACHE PATH "Directory containing the built wickra-copilot C ABI library")

set(WICKRA_COPILOT_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../bindings/c/include")

find_library(WICKRA_COPILOT_LIB
    NAMES wickra_copilot libwickra_copilot
    HINTS "${WICKRA_COPILOT_LIB_DIR}"
    REQUIRED)

add_executable(context context.c)
target_include_directories(context PRIVATE "${WICKRA_COPILOT_INCLUDE}")
target_link_libraries(context PRIVATE "${WICKRA_COPILOT_LIB}")

add_executable(contextpp context.cpp)
target_include_directories(contextpp PRIVATE "${WICKRA_COPILOT_INCLUDE}")
target_link_libraries(contextpp PRIVATE "${WICKRA_COPILOT_LIB}")

# The golden corpus, globbed at configure time and written into a header, so the
# test covers a spec added to the corpus without editing the test.
# CONFIGURE_DEPENDS makes the build system re-glob when the directory changes.
set(GOLDEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../golden")
file(GLOB GOLDEN_SPEC_FILES CONFIGURE_DEPENDS "${GOLDEN_DIR}/specs/*.json")
set(GOLDEN_SPEC_ENTRIES "")
foreach(spec_file ${GOLDEN_SPEC_FILES})
    get_filename_component(spec_name "${spec_file}" NAME)
    string(APPEND GOLDEN_SPEC_ENTRIES "    \"${spec_name}\",
")
endforeach()
list(LENGTH GOLDEN_SPEC_FILES GOLDEN_SPEC_COUNT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/golden_specs.h"
"/* Generated by examples/c/CMakeLists.txt -- do not edit. */
#ifndef WICKRA_COPILOT_GOLDEN_SPECS_H
#define WICKRA_COPILOT_GOLDEN_SPECS_H
#include <stddef.h>
#define GOLDEN_DIR \"${GOLDEN_DIR}\"
static const char *GOLDEN_SPECS[] = {
${GOLDEN_SPEC_ENTRIES}    NULL
};
static const size_t GOLDEN_SPEC_COUNT = ${GOLDEN_SPEC_COUNT};
#endif
")

# The test, as opposed to the examples above: `context` and `contextpp` print a
# context and exit zero. This one asserts golden parity and operating-mode
# equivalence over the whole corpus.
add_executable(golden_test golden_test.c)
target_include_directories(golden_test PRIVATE
    "${WICKRA_COPILOT_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(golden_test PRIVATE "${WICKRA_COPILOT_LIB}")

# On Windows the runtime DLL must sit next to the executable (there is no rpath);
# without it the loader pops a modal "DLL not found" dialog that hangs ctest. Copy
# it next to each example after build. (Unix resolves the .so/.dylib via the
# linker's embedded path, so this is Windows-only.)
if(WIN32)
    foreach(example context contextpp golden_test)
        add_custom_command(TARGET ${example} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${WICKRA_COPILOT_LIB_DIR}/wickra_copilot.dll"
                "$<TARGET_FILE_DIR:${example}>"
            VERBATIM)
    endforeach()
endif()

enable_testing()
add_test(NAME context COMMAND context)
add_test(NAME contextpp COMMAND contextpp)
add_test(NAME golden_test COMMAND golden_test)
# A missing runtime dependency must fail fast, never block the runner for hours.
set_tests_properties(context contextpp golden_test PROPERTIES TIMEOUT 60)
